<body>
  <h2>🍔 家族美食票券包 🍕</h2>
  <div id="ticket-container">
    <p style="text-align: center; color: #666;">讀取票券中，請稍候...</p>
  </div>

  <script>
    // ⚠️【請務必將下方的 URL 替換成你的 GAS 網址】
    const GAS_URL = 'https://script.google.com/macros/s/你的專屬ID/exec';

    async function fetchTickets() {
      try {
        const res = await fetch(GAS_URL);
        const tickets = await res.json();
        const container = document.getElementById('ticket-container');
        container.innerHTML = '';

        if(tickets.length === 0) {
          container.innerHTML = '<p style="text-align:center; color:#666;">目前沒有可用的票券囉！</p>';
          return;
        }

        tickets.forEach(ticket => {
          const card = document.createElement('div');
          card.className = 'card';
          card.id = `ticket-${ticket.row}`;
          
          // 【修正 0 開頭消失問題】：強制將傳來的資料轉換為字串
          const safeBarcode = String(ticket.barcode);
          
          // 根據類型決定要準備 SVG (一維) 還是 DIV (QR)
          let codeContainer = ticket.type === 'QR' 
            ? `<div id="qr-${ticket.row}" class="qr-container"></div>`
            : `<svg id="barcode-${ticket.row}"></svg>`;
          
          card.innerHTML = `
            <div class="brand">${ticket.brand}</div>
            <div class="expire">到期日：${ticket.expireDate}</div>
            ${codeContainer}
            <button class="btn" onclick="useTicket(${ticket.row})">我用掉了！(點擊核銷)</button>
          `;
          container.appendChild(card);
          
          // 判斷並渲染對應的圖形
          if (ticket.type === 'QR') {
            new QRCode(document.getElementById(`qr-${ticket.row}`), {
              text: safeBarcode,
              width: 150,
              height: 150
            });
          } else {
            JsBarcode(`#barcode-${ticket.row}`, safeBarcode, {
              format: "CODE128",
              width: 2,
              height: 70,
              displayValue: true,
              fontSize: 18
            });
          }
        });
      } catch (error) {
        document.getElementById('ticket-container').innerHTML = '<p style="text-align:center; color:red;">讀取失敗，請重新整理網頁。</p>';
      }
    }

    async function useTicket(rowNum) {
      if(!confirm('結帳刷完條碼了嗎？確定要核銷？')) return;
      
      const btn = document.querySelector(`#ticket-${rowNum} button`);
      btn.disabled = true;
      btn.innerText = '核銷處理中...';

      await fetch(GAS_URL, {
        method: 'POST',
        headers: { "Content-Type": "text/plain;charset=utf-8" },
        body: JSON.stringify({ row: rowNum })
      });

      document.getElementById(`ticket-${rowNum}`).style.display = 'none';
      alert('核銷成功！');
    }

    fetchTickets();
  </script>
